import { useState, useMemo, useEffect, useCallback, useRef } from 'react';
import { Server } from '@/types';
import { IconPlay, IconRefresh } from '@/components/icons';

// ─── Tipos internos del reproductor ───
type EmbedProvider = {
  name: string;
  baseUrl: string;
  sandbox?: string | false;
};

type PlayerServer = Server & {
  sandbox?: string | false;
};

// ─── Sandbox por defecto ───
// Este es el sandbox estándar para servidores seguros.
const DEFAULT_SANDBOX_PERMISSIONS = [
  'allow-scripts',
  'allow-same-origin',
  'allow-popups',
  'allow-presentation',
  'allow-forms',
].join(' ');

// ─── Sandbox menos restrictivo para VidLink ───
// Si VidLink sigue pidiendo deshabilitar sandbox, más abajo puedes cambiar
// sandbox: VIDLINK_SANDBOX_PERMISSIONS por sandbox: false.
const VIDLINK_SANDBOX_PERMISSIONS = [
  'allow-scripts',
  'allow-same-origin',
  'allow-popups',
  'allow-popups-to-escape-sandbox',
  'allow-forms',
  'allow-modals',
  'allow-presentation',
].join(' ');

// ─── Configuración de proveedores ───
const EMBED_PROVIDERS: EmbedProvider[] = [
  {
    name: 'VidSrc',
    baseUrl: 'https://vidsrc.to/embed/movie/',
    sandbox: DEFAULT_SANDBOX_PERMISSIONS,
  },

    baseUrl: 'https://vidlink.pro/movie/',
    sandbox: false,

    // Si VidLink no funciona con el sandbox menos restrictivo,
    // comenta la línea anterior y descomenta esta:
    // sandbox: false,
  },
  {
    name: 'VidEasy',
    baseUrl: 'https://player.videasy.to/movie/',
    sandbox: false,
  },
/*  {
    name: 'WatchOut',
    baseUrl: 'https://watchout-player.netlify.app/movie/',

*/  
  {

  },
  {
    name: 'Mapple',
    baseUrl: 'https://mapple.uk/watch/movie/',
    sandbox: false,
  },
  {
    name: 'VidZee',
    baseUrl: 'https://player.vidzee.wtf/embed/movie/',
    sandbox: false,
  },
];

// Tiempo máximo (ms) que esperamos a que un servidor responda antes de considerar que falló
const LOAD_TIMEOUT_MS = 15_000;

type LoadStatus = 'idle' | 'loading' | 'loaded' | 'timeout';

interface VideoPlayerProps {
  servers: Server[];
  tmdbId: number;
  movieSlug: string;
}

// ─── Utilidad: encontrar el siguiente servidor disponible ───
function findNextAvailableServer(
  servers: Server[],
  currentIndex: number,
  failedSet: Set<number>
): number {
  const total = servers.length;

  for (let i = 1; i < total; i++) {
    const candidateIndex = (currentIndex + i) % total;
    if (!failedSet.has(candidateIndex)) {
      return candidateIndex;
    }
  }

  return -1; // Todos fallaron
}

export default function VideoPlayer({
  servers,
  tmdbId,
  movieSlug,
}: VideoPlayerProps) {
  // ─── Lista combinada de servidores (dinámicos + manuales) ───
  const allServers = useMemo<PlayerServer[]>(() => {
    if (!tmdbId) return (servers || []) as PlayerServer[];

    const dynamicServers: PlayerServer[] = EMBED_PROVIDERS.map((provider) => ({
      id: `dynamic-${provider.name}`,
      name: provider.name,
      url: `${provider.baseUrl}${tmdbId}`,
      sandbox: provider.sandbox,
    }));

    return [...dynamicServers, ...(servers || []) as PlayerServer[]];
  }, [tmdbId, servers]);

  // ─── Estado del reproductor ───
  const [activeIndex, setActiveIndex] = useState(0);
  const [loadStatus, setLoadStatus] = useState<LoadStatus>('idle');
  const [autoSwitched, setAutoSwitched] = useState(false);
  const [retryCounter, setRetryCounter] = useState(0);

  // Estado real para UI
  const [failedServers, setFailedServers] = useState<Set<number>>(
    () => new Set<number>()
  );

  // Ref auxiliar para lógica interna sin problemas de stale state
  const failedServersRef = useRef<Set<number>>(failedServers);
  const timeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  const activeServer = allServers[activeIndex] ?? null;

  // Mantener el ref sincronizado con el estado
  useEffect(() => {
    failedServersRef.current = failedServers;
  }, [failedServers]);

  // ─── Limpiar timeout ───
  const clearLoadTimeout = useCallback(() => {
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
      timeoutRef.current = null;
    }
  }, []);

  // ─── Helpers para mutar failedServers de forma segura ───
  const markServerFailed = useCallback((index: number) => {
    const next = new Set(failedServersRef.current);
    next.add(index);
    failedServersRef.current = next;
    setFailedServers(next);
  }, []);

  const clearServerFailed = useCallback((index: number) => {
    const next = new Set(failedServersRef.current);
    next.delete(index);
    failedServersRef.current = next;
    setFailedServers(next);
  }, []);

  const resetFailedServers = useCallback(() => {
    const next = new Set<number>();
    failedServersRef.current = next;
    setFailedServers(next);
  }, []);

  // ─── Protección por si la lista de servidores cambia y el índice queda fuera de rango ───
  useEffect(() => {
    if (allServers.length === 0) return;

    if (activeIndex >= allServers.length) {
      setActiveIndex(0);
    }
  }, [allServers.length, activeIndex]);

  // ─── Iniciar timer de carga cuando cambia el servidor activo ───
  useEffect(() => {
    clearLoadTimeout();

    if (!activeServer) {
      setLoadStatus('idle');
      return;
    }

    setLoadStatus('loading');

    timeoutRef.current = setTimeout(() => {
      markServerFailed(activeIndex);

      const nextIndex = findNextAvailableServer(
        allServers,
        activeIndex,
        failedServersRef.current
      );

      if (nextIndex !== -1) {
        setAutoSwitched(true);
        setActiveIndex(nextIndex);
      } else {
        setLoadStatus('timeout');
      }
    }, LOAD_TIMEOUT_MS);

    return clearLoadTimeout;
  }, [
    activeIndex,
    activeServer,
    allServers,
    clearLoadTimeout,
    markServerFailed,
    retryCounter,
  ]);

  // ─── Resetear estado cuando cambia la película ───
  useEffect(() => {
    setActiveIndex(0);
    setLoadStatus('idle');
    setAutoSwitched(false);
    setRetryCounter(0);
    resetFailedServers();
    clearLoadTimeout();
  }, [movieSlug, clearLoadTimeout, resetFailedServers]);

  // ─── El iframe terminó de cargar ───
  const handleIframeLoad = useCallback(() => {
    clearLoadTimeout();
    setLoadStatus('loaded');
  }, [clearLoadTimeout]);

  // ─── Reintentar el servidor actual ───
  const handleRetry = useCallback(() => {
    clearServerFailed(activeIndex);
    setAutoSwitched(false);
    setLoadStatus('loading');
    setRetryCounter((prev) => prev + 1);
  }, [activeIndex, clearServerFailed]);

  // ─── Cambiar manualmente de servidor ───
  const handleSelectServer = useCallback(
    (index: number) => {
      clearServerFailed(index);
      setAutoSwitched(false);

      if (index === activeIndex) {
        // Si el usuario hace clic en el mismo servidor, forzamos remount/reintento
        setLoadStatus('loading');
        setRetryCounter((prev) => prev + 1);
      } else {
        setActiveIndex(index);
      }
    },
    [activeIndex, clearServerFailed]
  );

  // ─── Si no hay servidores ───
  if (!allServers || allServers.length === 0) {
    return (
      <div className="aspect-video bg-white/[0.02] border border-white/5 rounded-2xl flex items-center justify-center text-gray-500">
        No hay servidores disponibles para esta película.
      </div>
    );
  }

  const allFailed = failedServers.size >= allServers.length;

  // ─── Sandbox final del iframe ───
  // Si activeServer.sandbox === false, se omite el atributo sandbox.
  // Si no tiene sandbox definido, se usa DEFAULT_SANDBOX_PERMISSIONS.
  const iframeSandbox =
    activeServer?.sandbox === false
      ? undefined
      : (activeServer?.sandbox ?? DEFAULT_SANDBOX_PERMISSIONS);

  return (
    <div className="w-full">
      {/* ═══ Selector de servidores ═══ */}
      <div className="flex flex-wrap items-center gap-2 mb-4">
        {allServers.map((server, index) => {
          const isActive = index === activeIndex;
          const isFailed = failedServers.has(index);

          return (
            <button
              key={
                typeof server.id === 'string'
                  ? server.id
                  : `${server.id}-${index}`
              }
              onClick={() => handleSelectServer(index)}
              aria-pressed={isActive}
              title={
                isFailed
                  ? 'Este servidor no respondió. Clic para reintentar.'
                  : server.name
              }
              className={`inline-flex items-center gap-1.5 px-4 py-2 rounded-full text-sm font-medium transition-all duration-200 border ${
                isActive
                  ? 'bg-red-600/80 border-red-500/30 text-white shadow-lg shadow-red-900/20'
                  : isFailed
                    ? 'bg-white/[0.03] border-white/5 text-gray-600 hover:bg-white/5 hover:text-gray-400'
                    : 'bg-white/5 border-white/10 text-gray-300 hover:bg-white/10 hover:border-white/20 hover:text-white'
              }`}
            >
              {isActive && <IconPlay className="w-3 h-3" />}
              {server.name}
              {isFailed && !isActive && (
                <span className="text-[9px] uppercase tracking-wide opacity-60">
                  caído
                </span>
              )}
            </button>
          );
        })}
      </div>

      {/* ═══ Notificación de cambio automático ═══ */}
      {autoSwitched && loadStatus === 'loading' && (
        <div className="mb-3 flex items-center gap-2 text-xs text-amber-400/80 bg-amber-500/5 border border-amber-500/10 rounded-lg px-3 py-2">
          <span className="w-1.5 h-1.5 bg-amber-400 rounded-full animate-pulse flex-shrink-0" />
          El servidor anterior no respondió. Cambiando automáticamente a{' '}
          <strong>{activeServer?.name}</strong>...
        </div>
      )}

      {/* ═══ Contenedor del video ═══ */}
      <div className="relative aspect-video bg-black rounded-2xl overflow-hidden shadow-2xl shadow-black/30 border border-white/5">
        {/* ─── Overlay de carga ─── */}
        {loadStatus === 'loading' && (
          <div className="absolute inset-0 z-10 flex flex-col items-center justify-center gap-3 bg-black/80">
            <div className="w-10 h-10 border-4 border-red-600/60 border-t-transparent rounded-full animate-spin" />
            <p className="text-sm text-gray-400">
              Conectando con{' '}
              <span className="text-white font-medium">
                {activeServer?.name}

interface DropdownMenuProps {
  id: string;
  icon: ReactNode;
  label: string;
  currentLabel: string;
  isActive: boolean;
  align?: 'left' | 'right';
  closeOnSelect?: boolean;
  children: ReactNode;
}

function DropdownMenu({
  id,
  icon,
  label,
  currentLabel,
  isActive,
  align = 'left',
  closeOnSelect = true,
  children,
}: DropdownMenuProps) {
  const [isOpen, setIsOpen] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const handleClickOutside = (event: MouseEvent) => {
      if (ref.current && !ref.current.contains(event.target as Node)) {
        setIsOpen(false);
      }
    };
    const handleEscape = (event: KeyboardEvent) => {
      if (event.key === 'Escape') setIsOpen(false);
    };

    if (isOpen) {
      document.addEventListener('mousedown', handleClickOutside);
      document.addEventListener('keydown', handleEscape);
    }
    return () => {
      document.removeEventListener('mousedown', handleClickOutside);
      document.removeEventListener('keydown', handleEscape);
    };
  }, [isOpen]);

  return (
    <div className="relative" ref={ref}>
      <button
        onClick={() => setIsOpen(!isOpen)}
        aria-expanded={isOpen}
        aria-controls={`${id}-dropdown`}
        aria-label={`${label}: ${currentLabel}`}
        title={`${label}: ${currentLabel}`}
        className={`flex items-center gap-2 px-4 py-2 rounded-full text-sm font-medium transition-all duration-200 border ${isActive
            ? 'bg-red-600/80 border-red-500/30 text-white shadow-lg shadow-red-900/20'
            : 'bg-white/5 border-white/10 text-gray-300 hover:bg-white/10 hover:border-white/20 hover:text-white'
          }`}
      >
        <span className={isActive ? 'text-red-100' : 'text-gray-500'}>{icon}</span>
        <span className="hidden sm:inline">{currentLabel}</span>
        <IconChevron open={isOpen} className="w-3 h-3 opacity-60" />
      </button>

      {isOpen && (
        <div
          id={`${id}-dropdown`}
          className={`absolute z-[60] top-full ${align === 'right' ? 'right-0' : 'left-0'} min-w-[200px] pt-2`}
          onClick={(e) => {
            if (closeOnSelect) setIsOpen(false);
            else e.stopPropagation();
          }}
        >
          {children}
        </div>
      )}
    </div>
  );
}

// lib/api/countries.ts

import { fetchApi } from './http';

export async function fetchCountries(): Promise<string[]> {
  const json = await fetchApi<unknown>('/movies/countries', {}, {
    next: {
      revalidate: 3600,
      tags: ['countries'],
    },
  });

  const rawList = Array.isArray(json)
    ? json
    : Array.isArray((json as { data?: unknown[] })?.data)
      ? (json as { data: unknown[] }).data
      : [];

  const names = rawList
    .map((item) => {
      if (typeof item === 'string') return item;

      if (item && typeof item === 'object') {
        const obj = item as Record<string, unknown>;
        return (obj.name ?? obj.country ?? obj.value) as string | undefined;
      }

      return undefined;
    })
    .filter((value): value is string => Boolean(value && value.trim()));

  const unique = Array.from(new Set(names.map((name) => name.trim())));

  return endpoint.replace(/^\/+/, '');

export function buildApiUrl(
  endpoint: string,
  params: Record<string, string | string[] | undefined>
) {
  const base = cleanApiUrl(API_URL);
  const path = cleanEndpoint(endpoint);
  const url = new URL(`${base}/api/${path}`);
  Object.entries(params).forEach(([key, value]) => {
    if (value === undefined || value === null || value === '') return;
    if (Array.isArray(value)) {
      value.forEach((item) => {
        if (item) url.searchParams.append(key, item);
      });
    } else {
      url.searchParams.set(key, value);
    }
  });
  return url;
}

export async function fetchApi<TJson = unknown>(
  endpoint: string,
  params: Record<string, string | string[] | undefined> = {},
  options: NextFetchRequestConfig = {}
): Promise<TJson | null> {
  const url = buildApiUrl(endpoint, params);
  const headers = new Headers(options.headers);

  if (options.body && !headers.has('Content-Type')) {
    headers.set('Content-Type', 'application/json');
  }

  // Timeout con AbortController
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), FETCH_TIMEOUT_MS);

  try {
    const res = await fetch(url.toString(), {
      ...options,
      headers,
      signal: controller.signal,
    });

    clearTimeout(timeout);

    if (!res.ok) {
      let errorBody = '';
      try {
        errorBody = await res.text();
      } catch {
        // Si no podemos leer el body, continuamos
      }
      console.error(`API Error [${res.status}]: ${url.toString()}`);
      if (errorBody) {
        console.error('Error details:', errorBody);
      }
      return null;
    }

    return (await res.json()) as TJson;
  } catch (error) {
    clearTimeout(timeout);

    if (error instanceof DOMException && error.name === 'AbortError') {
      console.error(`Timeout: Strapi no respondió en ${FETCH_TIMEOUT_MS / 1000}s → ${url.toString()}`);
    } else {
      console.error(`Network Error fetching ${url.toString()}:`, error);
    }
    return null;
  }
}

interface IndexEntry {

/**

 */
export function normalizeTerm(value: string): string {
  return value
    .normalize('NFD')
    .replace(/\p{Diacritic}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

/**
 * Distancia de Levenshtein con límite (early-exit).
 * Devuelve la distancia si es <= max, o max+1 si la supera.
 */
function levenshteinBounded(a: string, b: string, max: number): number {
  const lenA = a.length;
  const lenB = b.length;
  if (Math.abs(lenA - lenB) > max) return max + 1;
  if (lenA === 0) return lenB <= max ? lenB : max + 1;
  if (lenB === 0) return lenA <= max ? lenA : max + 1;

  let prev: number[] = Array.from({ length: lenB + 1 }, (_, j) => j);

  for (let i = 1; i <= lenA; i++) {
    const curr: number[] = [i];
    let rowMin = i;
    for (let j = 1; j <= lenB; j++) {
      const cost = a[i - 1] === b[j - 1] ? 0 : 1;
      const val = Math.min(prev[j] + 1, curr[j - 1] + 1, prev[j - 1] + cost);
      curr.push(val);
      if (val < rowMin) rowMin = val;
    }
    if (rowMin > max) return max + 1;
    prev = curr;
  }

  return prev[lenB] <= max ? prev[lenB] : max + 1;
}

/**
 * Devuelve hasta `limit` sugerencias para una query mal escrita.
 * Entiende: typos ("inseption"), tildes ("mision"), prefijos ("dicapr").
 */
export async function getSuggestions(query: string, limit = 3): Promise<string[]> {
  const normQuery = normalizeTerm(query);
  if (normQuery.length < 3) return [];

  const index = await getSearchIndex();
  const scored: { display: string; score: number }[] = [];

  for (const entry of index) {
    let best = Number.POSITIVE_INFINITY;

    // 1) Coincidencia exacta o prefijo del nombre completo
    if (entry.norm === normQuery) {
      best = 0;
    } else if (entry.norm.startsWith(normQuery)) {
      best = 1;
    } else {
      // 2) Comparación palabra por palabra
      for (const token of entry.tokens) {
        if (token === normQuery) { best = 0; break; }

        if (normQuery.length >= 4 && token.startsWith(normQuery)) {
          if (best > 1) best = 1;
          continue;
        }

        const maxDist = normQuery.length <= 5 ? 1 : 2;
        if (Math.abs(token.length - normQuery.length) > maxDist) continue;

        const d = levenshteinBounded(normQuery, token, maxDist);
        if (d < best) best = d;
        if (best === 0) break;
      }

      // 3) Para queries de varias palabras, comparar contra el nombre completo
      if (best > 2 && normQuery.includes(' ')) {
        const d = levenshteinBounded(normQuery, entry.norm, 2);
        if (d < best) best = d;
      }
    }

    if (best <= 2) scored.push({ display: entry.display, score: best });
  }

  scored.sort((a, b) => a.score - b.score || a.display.localeCompare(b.display));
  return scored.slice(0, limit).map((s) => s.display);
}

  const posterUrl = getMediaUrl(movie.poster, 'medium') ?? movie.poster_url ?? undefined;
  const backdropUrl = getMediaUrl(movie.backdrop ?? movie.poster, 'large') ?? movie.backdrop_url ?? undefined;

  // ═══ SLUG DE FALLBACK ═══
  // Si Strapi no pudo generar un slug (títulos en escritura no latina),
  // generamos uno usando el título transliterado o el tmdb_id como último recurso.
  let slug = movie.slug ?? null;
  if (!slug) {
    const transliterated = slugify(movie.title);
    if (transliterated && transliterated.length >= 3) {
      slug = `${transliterated}-${movie.id}`;
    } else {
      slug = `tmdb-${movie.tmdb_id ?? movie.id}`;
    }
  }

  return {
    id: movie.id,
    documentId: movie.documentId ?? String(movie.id),
    title: movie.title,
    english_title: movie.english_title ?? undefined,  // ← NUEVO
    spanish_title: movie.spanish_title ?? undefined,
    slug,  // ← usamos el slug garantizado
    synopsis: movie.synopsis ?? '',
    year: movie.year ?? 0,
    rating: movie.rating ?? 0,
    country: movie.country ?? '',
    tmdb_id: movie.tmdb_id ?? 0,
    views: movie.views ?? undefined,
    featured_this_week: movie.is_featured ?? undefined,
    poster_url: posterUrl,
    backdrop_url: backdropUrl,
    trailer_url: movie.trailer_url ?? undefined,
    runtime: movie.runtime ?? undefined,
    age_rating: movie.age_rating ?? undefined,
    original_language: movie.original_language ?? undefined,
    servers,
    categories,
    actors,
    directors,
  };
}

export function normalizeMovieList(json: StrapiListResponse<unknown> | null): Movie[] {
  if (!json?.data || !Array.isArray(json.data)) return [];

    .filter((movie): movie is Movie => movie !== null);
}

function computeSearchScore(movie: Movie, normQuery: string, source: SearchSource): number {
  const SOURCE_BASE: Record<SearchSource, number> = {
    title: 70,      // ← título siempre gana
    actor: 45,
    director: 40,
    category: 35,
  };

  let score = SOURCE_BASE[source];

  if (source === 'title') {
    const candidates = [movie.title, movie.english_title, movie.spanish_title]
      .filter((t): t is string => Boolean(t && typeof t === 'string'));

    let best = 0;
    for (const raw of candidates) {
      const nt = normalizeSearchTerm(raw);
      if (nt === normQuery) best = Math.max(best, 30);           // coincidencia exacta
      else if (nt.startsWith(normQuery)) best = Math.max(best, 20); // empieza con
      else if (nt.includes(normQuery)) best = Math.max(best, 10);   // contiene
      else {
        // coincidencia parcial por palabras
        const qWords = normQuery.split(' ').filter(w => w.length > 1);
        const tWords = nt.split(' ');
        const matched = qWords.filter(qw => tWords.some(tw => tw.includes(qw)));
        if (matched.length > 0) best = Math.max(best, 5 + matched.length * 2);
      }
    }
    score += best;
  } else if (source === 'actor') {
    const exact = (movie.actors || []).some(a => normalizeSearchTerm(a.name) === normQuery);
    if (exact) score += 10;
  } else if (source === 'director') {
    const exact = (movie.directors || []).some(d => normalizeSearchTerm(d.name) === normQuery);
    if (exact) score += 10;
  } else if (source === 'category') {
    const exact = (movie.categories || []).some(c => normalizeSearchTerm(c.name) === normQuery);
    if (exact) score += 10;
  }

  return score;
}

export async function searchMovies(
  query: string,
  page = 1,
  pageSize = SEARCH_PAGE_SIZE
): Promise<PaginatedResult<Movie>> {
  const safeQuery = query.trim();
  const safePage = Math.max(1, Math.floor(Number(page) || 1));
  const safePageSize = Math.max(1, Math.floor(Number(pageSize) || SEARCH_PAGE_SIZE));

  if (!safeQuery) {
    return { data: [], hasMore: false, page: safePage, pageSize: safePageSize };
  }

  const normQuery = normalizeSearchTerm(safeQuery);

  // Límites por sub-búsqueda (ajustables)
  const titleLimit = Math.min(Math.max(safePageSize * 4, 80), 200);
  const otherLimit = Math.min(Math.max(safePageSize * 2, 40), 100);

  // ─── 4 búsquedas en paralelo, cada una con su fuente ───
  const [titleMovies, actorMovies, directorMovies, categoryMovies] = await Promise.all([
    searchMoviesByTitle(safeQuery, titleLimit),
    searchMoviesByActorName(safeQuery, otherLimit),
    searchMoviesByDirectorName(safeQuery, otherLimit),
    searchMoviesByCategoryName(safeQuery, otherLimit),
  ]);

  // ─── Merge + scoring + dedup por documentId ───
  const mergedMap = new Map<string, { movie: Movie; score: number }>();

  const addResults = (movies: Movie[], source: SearchSource) => {
    for (const movie of movies) {
      const key = movie.documentId || String(movie.id);
      const score = computeSearchScore(movie, normQuery, source);
      const existing = mergedMap.get(key);
      // Nos quedamos con el score más alto si la película aparece en varias fuentes
      if (!existing || score > existing.score) {
        mergedMap.set(key, { movie, score });
      }
    }
  };

  addResults(titleMovies, 'title');
  addResults(actorMovies, 'actor');
  addResults(directorMovies, 'director');
  addResults(categoryMovies, 'category');

  // ─── Ordenar por relevancia ───
  const sorted = Array.from(mergedMap.values()).sort((a, b) => {
    if (b.score !== a.score) return b.score - a.score;
    const ra = a.movie.rating ?? 0;
    const rb = b.movie.rating ?? 0;
    if (rb !== ra) return rb - ra;
    const ya = a.movie.year ?? 0;
    const yb = b.movie.year ?? 0;
    if (yb !== ya) return yb - ya;
    return a.movie.title.localeCompare(b.movie.title);
  });

  // ─── Paginar ───
  const start = (safePage - 1) * safePageSize;
  const paged = sorted.slice(start, start + safePageSize).map(entry => entry.movie);
  const hasMore = start + safePageSize < sorted.length;

  if (process.env.NODE_ENV !== 'production') {
    console.log(`[searchMovies] "${safeQuery}" → ${sorted.length} resultados relevantes`);
  }

  return { data: paged, hasMore, page: safePage, pageSize: safePageSize };
}

      pageSize,

import { searchMovies } from '@/lib/api';
import AdBanner from '@/components/AdBanner';
import MovieCard from '@/components/MovieCard';
import Link from 'next/link';

export default async function SearchPage({
  searchParams,
}: {
  searchParams: Promise<{ q?: string }>;
}) {
  const { q } = await searchParams;

  // ─── Sin query: estado inicial ───
  if (!q || !q.trim()) {
    return (
      <main className="min-h-screen p-8 text-center">
        <p className="text-gray-400 mt-20">
          Escribe algo en la barra de búsqueda...
        </p>
      </main>
    );
  }

  const query = q.trim();

  // ─── Búsqueda principal ───
  let result = await searchMovies(query);
  const count = result.data.length;

  return (
    <main className="min-h-screen p-8">
      {/* ─── Encabezado ─── */}
      <div className="mb-6">
        <h2 className="text-2xl font-bold text-white">
          Buscaste: <span className="text-red-500">"{query}"</span>
        </h2>
        <p className="text-sm text-gray-400 mt-1">
          {count === 0
            ? 'Sin coincidencias en el catálogo'
            : `${count} ${count === 1 ? 'resultado' : 'resultados'}`}
        </p>
      </div>

      {/* ─── Resultados ─── */}
      {count === 0 ? (
        <div className="flex flex-col items-center justify-center py-24 text-center">
          <h3 className="text-lg font-semibold text-gray-200 mb-1">Sin resultados</h3>
          <p className="text-sm text-gray-400 mb-6 max-w-sm">

// app/sitemap.ts
import type { MetadataRoute } from 'next';
import { fetchMoviesWithFilters } from '@/lib/api';

const BASE_URL = process.env.NEXT_PUBLIC_SITE_URL || 'https://tudominio.com';

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  // Rutas estáticas
  const staticRoutes: MetadataRoute.Sitemap = [
    {
      url: BASE_URL,
      lastModified: new Date(),
      changeFrequency: 'daily',
      priority: 1.0,
    },
    {
      url: `${BASE_URL}/movies`,
      lastModified: new Date(),
      changeFrequency: 'daily',
      priority: 0.9,
    },
  ];

  // Películas: obtener slugs del catálogo
  // Nota: para catálogos muy grandes, considera paginar aquí
  const movieRoutes: MetadataRoute.Sitemap = [];
  let page = 1;
  let hasMore = true;

  while (hasMore && page <= 50) { // Límite de seguridad: 50 páginas × 100 = 5000 películas
    try {
      const result = await fetchMoviesWithFilters({}, page, 100);
      const movies = result.data;

      movies.forEach((movie) => {
        movieRoutes.push({
          url: `${BASE_URL}/movie/${movie.slug}`,
          lastModified: new Date(),
          changeFrequency: 'weekly',
          priority: 0.8,
        });
      });

      hasMore = result.hasMore;
      page++;
    } catch {
      break;
    }
  }

  return [...staticRoutes, ...movieRoutes];
}

    key: 'X-Frame-Options',
    value: 'DENY',
  },
  {
    // Activa el filtro XSS básico del navegador (legacy, pero no hace daño tenerlo).
    key: 'X-XSS-Protection',
    value: '1; mode=block',
  },

    // "strict-origin-when-cross-origin" es el balance correcto:
